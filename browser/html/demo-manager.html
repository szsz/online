Document Manager (Demo)

Uses the same postMessage protocol as the webclient Content Preview service. Viewer is served via reverse proxy at /viewer/* (same-origin for SharedArrayBuffer).

Required server setup: 1. Send COOP: same-origin + COEP: require-corp headers. 2. Reverse-proxy /viewer/* to the COOL viewer server.
Select a file to open in the embedded editor.

Log

Integration code:

// Same message protocol as webclient Content Preview service.
// Viewer served same-origin via reverse proxy (/viewer/* → COOL server).

var viewer = document.getElementById('viewer');

window.addEventListener('message', function(e) {
    switch (e.data.type) {
        case 'event-listener-ready':
            // Viewer ready — send document
            viewer.contentWindow.postMessage({
                type: 'view-document',
                payload: {
                    blob: fileBlob,         // Blob
                    fileName: 'report.docx',
                    language: 'en',
                }
            }, window.location.origin);
            break;

        case 'document-loaded':
            console.log('Document rendered');
            break;

        case 'save-docx-content':
            // e.data.payload.payload.blob = saved Blob
            download(e.data.payload.payload.blob);
            break;

        case 'error':
            console.error('Viewer error:', e.data.payload);
            break;
    }
});

// Request save
viewer.contentWindow.postMessage(
    { type: 'save-docx' },
    window.location.origin
);
